% Pandoc's template may load geometry before this header file. Use
% \geometry{...} to override the existing settings instead of trying to
% reload the package with options.
\usepackage{geometry}
\usepackage{etoolbox}

% Memoir defaults to letter stock. Force both stock and trim to 6x9 so the
% PDF page size is actually 6x9 instead of a trimmed 6x9 on letter stock.
\setstocksize{9in}{6in}
\settrimmedsize{9in}{6in}{*}
\settrims{0pt}{0pt}
\geometry{
  paperwidth=6in,
  paperheight=9in,
  left=0.6in,
  right=0.6in,
  top=0.5in,
  bottom=0.85in,
  heightrounded
}

% Belt and suspenders: explicitly set the PDF page size for XeTeX.
\AtBeginDocument{%
  \special{papersize=6in,9in}%
  \setlength{\pdfpagewidth}{6in}%
  \setlength{\pdfpageheight}{9in}%
}

\raggedbottom

% Switching to memoir
\setlength{\parindent}{0pt}
\setlength{\parskip}{1em}
\renewcommand{\chapnamefont}{}
\renewcommand{\printchaptername}{}
\renewcommand{\chapnumfont}{}
\renewcommand{\printchapternum}{}
\setlength{\beforechapskip}{0em}
\setlength{\afterchapskip}{2em}
\setsecnumdepth{none}

% ==== Force format for \section*{Contents} ====
\usepackage{titlesec}

% ==== Redefine \tableofcontents with a small "Contents" header ====
\makeatletter
\renewcommand\tableofcontents{%
  \begingroup
    \setlength{\parskip}{1pt}%
    \setlength{\parindent}{0pt}%
    \vspace{3em}%
    {\LARGE \bfseries Contents\par}
    \vspace{0.5em}
    \@starttoc{toc}%
  \endgroup
}
\makeatother

% ##
\titleformat{\chapter}
  {\normalfont\Huge\bfseries\raggedright}{\thechapter}{1em}{}
\titleformat{\section}
  {\normalfont\Large\bfseries\raggedright}{\thesection}{1em}{}
\titleformat{\subsection}
  {\normalfont\large\bfseries\raggedright}{\thesubsection}{1em}{}
\titlespacing*{\section}{0pt}{0.8em}{0.1em}
\titlespacing*{\subsection}{0pt}{0.7em}{0em}
\titleformat{\subsubsection}
  {\normalfont\normalsize\bfseries\raggedright}{\thesubsubsection}{1em}{}
\titlespacing*{\subsubsection}{0pt}{0.5em}{0em}
\titleformat{\paragraph}
  {\normalfont\normalsize\bfseries\raggedright}{\theparagraph}{1em}{}
\titlespacing*{\paragraph}{0pt}{0.3em}{0em}
\titleformat{\subparagraph}
  {\normalfont\normalsize\bfseries\itshape\raggedright}{\thesubparagraph}{1em}{}
\titlespacing*{\subparagraph}{0pt}{0.2em}{0em}
